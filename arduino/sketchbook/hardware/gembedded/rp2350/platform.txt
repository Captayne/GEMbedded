# Copyright (C) 2026 Andreas Keibel
#
# A sketch here is an ordinary GEM program: it has its own main(), calls
# appl_init() and runs its own event loop.  Nothing of Arduino's own
# framework is linked in -- no main(), no setup(), no loop() -- so the
# program looks exactly like one written on an Atari, and every AES and
# VDI call is available (cores/gem).  The second half of a program, the
# one on the real-time core, comes from irk.h.
#
# Everything this needs lives inside the GEMbedded folder: the toolchain
# in arduino/toolchain (tools/get-toolchain.py puts it there), the C
# library in lib/, the headers in pTOS/.  Nothing is installed anywhere.

name=GEMbedded
version=0.1.0

# How the IDE finds the machine and talks to it.  Without these it does
# not connect the port you chose with this board, and an upload fails
# before anything of ours is called.
pluggable_discovery.required.0=builtin:serial-discovery
pluggable_discovery.required.1=builtin:mdns-discovery
pluggable_monitor.required.serial=builtin:serial-monitor

# ---- where things are -------------------------------------------------
# runtime.platform.path is .../arduino/sketchbook/hardware/gembedded/rp2350
gembedded.root={runtime.platform.path}/../../../../..
compiler.path={gembedded.root}/arduino/toolchain/bin/
gembedded.tools={gembedded.root}/tools
gembedded.lib={runtime.platform.path}/lib
gembedded.ptos={gembedded.root}/pTOS

compiler.prefix=arm-none-eabi-

# ---- flags ------------------------------------------------------------
# softfp: the machine has a single-precision unit and programs may use it,
# but arguments travel in the integer registers, the way pTOS and libcmini
# are built.
compiler.cpu=-mcpu=cortex-m33 -mthumb -mfloat-abi=softfp -mfpu=fpv5-sp-d16
compiler.warning_flags=-Wall
compiler.warning_flags.none=-w
compiler.warning_flags.default=-Wall
compiler.warning_flags.more=-Wall -Wextra
compiler.warning_flags.all=-Wall -Wextra -Wpedantic

compiler.includes=-I{gembedded.ptos}/lib/libcmini/include -iquote {gembedded.ptos}/include -iquote {build.core.path}
compiler.common=-O2 -g -mword-relocations -ffunction-sections -fdata-sections {compiler.includes}

compiler.c.cmd={compiler.prefix}gcc
compiler.c.flags=-c {compiler.common} {compiler.warning_flags}
compiler.cpp.cmd={compiler.prefix}g++
compiler.cpp.flags=-c {compiler.common} -fno-exceptions -fno-rtti -fno-threadsafe-statics {compiler.warning_flags}
compiler.S.cmd={compiler.prefix}gcc
compiler.S.flags=-c -x assembler-with-cpp {compiler.includes}
compiler.ar.cmd={compiler.prefix}ar
compiler.ar.flags=rcs
compiler.c.elf.cmd={compiler.prefix}gcc
# --target1-abs: the table of global constructors is relocated with
# R_ARM_TARGET1, which the loader of pTOS does not know; as ABS32 it
# is the same relocation under a name it does.
# --wrap=main: global constructors run before the program starts
# (cores/gem/cxx.cpp); libcmini's start-up does not do it.
# -u __wrap_main: and that wrapper has to be linked in for it to happen.
# The core archive is named before -lcmini, and at that point nothing has
# asked for __wrap_main yet -- the reference appears later, when
# libcmini's minimain.o is pulled in, and the linker does not go back for
# an archive it has already passed.  Entering the symbol as undefined
# before the link starts makes cxx.o come out on the first sweep.
compiler.c.elf.flags=-nostartfiles -nostdlib -Wl,-q -Wl,-Ttext=0 -Wl,-e_start -Wl,--wrap=main -Wl,-u,__wrap_main -Wl,--target1-abs
compiler.libraries.ldflags=
compiler.objcopy.cmd={compiler.prefix}objcopy
compiler.size.cmd={compiler.prefix}size

# ---- recipes ----------------------------------------------------------
recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.cpu} {compiler.c.flags} {includes} "{source_file}" -o "{object_file}"
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpu} {compiler.cpp.flags} {includes} "{source_file}" -o "{object_file}"
recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.cpu} {compiler.S.flags} {includes} "{source_file}" -o "{object_file}"
recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} "{archive_file_path}" "{object_file}"

# The program: the C library's start-up first, then the sketch, then what
# it uses.  pTOS loads ELF files, so the .PRG is the linker's output.
recipe.c.combine.pattern="{compiler.path}{compiler.c.elf.cmd}" {compiler.cpu} {compiler.c.elf.flags} "{gembedded.lib}/minicrt0.o" {object_files} "{build.path}/{archive_file}" "-L{gembedded.lib}" -lcmini -lm -lgcc -o "{build.path}/{build.project_name}.PRG"

# No objcopy step: pTOS loads ELF files, so what the linker wrote is
# already the program.  (A recipe.objcopy.* would run here if there
# were one -- there is nothing left to convert.)
recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -A "{build.path}/{build.project_name}.PRG"
recipe.size.regex=^(?:\.text|\.data|\.rodata)\s+([0-9]+).*

# what the IDE offers to export ("Sketch -> Export compiled binary")
recipe.output.tmp_file={build.project_name}.PRG
recipe.output.save_file={build.project_name}.PRG

# ---- upload -----------------------------------------------------------
# gemdeploy sends the program over the USB console; DEPLOY has to be
# running on the machine.  A resource file next to the sketch goes first,
# so that the program finds it when it starts.
# "python" on Windows is often the Store placeholder, which does nothing
# but open the Store, so a small launcher there picks the first
# interpreter that really answers.  Elsewhere python3 is python3.
tools.gemdeploy.path={gembedded.tools}
tools.gemdeploy.upload.params.verbose=
tools.gemdeploy.upload.params.quiet=
tools.gemdeploy.upload.pattern=python3 "{path}/gemdeploy.py" -p {serial.port} "{build.path}/{build.project_name}.PRG"
tools.gemdeploy.upload.pattern.windows="{path}/gemdeploy.bat" -p {serial.port} "{build.path}/{build.project_name}.PRG"
